Comandos que auxiliam na visualização do conteúdos de banco de dados grandes:

  
SELECT * FROM curso

ORDER BY instrutor DESC;

ou

order by nome_do_curso desc

Para Selecionar somente alguns campos para consulta:

  
select nome, carga from curso;

Selecionando com 'where'(enquanto):

  
select ano, nome from curso
where ano = 2020     
order by nome asc; 
order by nome desc; 

describe nome_do_curso

Selecionando linhas e colunas:

  
select nome, curso, ano from cursos
where ano <= 2022
order by nome desc;

Usando Between para selecionar:

  
select nome, ano from cursos
where ano between 2020 and 2024 
order by ano desc, nome asc;

Indicando faixa de valores:

  
select nome, curso, ano from curso
where ano in (2019, 2023, 2025)
order by ano asc;

Selecionando entre um valor e(and) outro:

  
select nome, carga, aulas from cursos
where carga > 35 and aulas < 30;

describe cursos;

Selecionando entre um valor e(or) outro:

  
select nome, carga, aulas from cursos
where carga > 35 or aulas < 30;

describe cursos;
